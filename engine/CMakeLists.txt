cmake_minimum_required(VERSION 3.16)
project(omori CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

file(GLOB_RECURSE ENGINE_SRC CONFIGURE_DEPENDS src/core/*.cpp src/js/*.cpp src/gfx2d/*.cpp src/rpg/*.cpp src/omori/*.cpp src/main.cpp)
list(APPEND ENGINE_SRC ${CMAKE_SOURCE_DIR}/../shared/rx.c ${CMAKE_SOURCE_DIR}/../shared/js.c ${CMAKE_SOURCE_DIR}/../shared/node.c ${CMAKE_SOURCE_DIR}/../shared/buf.c ${CMAKE_SOURCE_DIR}/../shared/script.c)

set(BUILD_INFO_DIR ${CMAKE_BINARY_DIR}/gen)
set(BUILD_INFO_ARGS -DSRC=${CMAKE_SOURCE_DIR}/.. -DOUT=${BUILD_INFO_DIR}/build_info.h -P ${CMAKE_SOURCE_DIR}/cmake/build_info.cmake)
execute_process(COMMAND ${CMAKE_COMMAND} ${BUILD_INFO_ARGS})
add_custom_target(build_info ALL COMMAND ${CMAKE_COMMAND} ${BUILD_INFO_ARGS} BYPRODUCTS ${BUILD_INFO_DIR}/build_info.h)

if(PSP)
  set(OMORI_TARGET psp)
elseif(NINTENDO_3DS)
  set(OMORI_TARGET 3ds)
elseif(NINTENDO_WII)
  set(OMORI_TARGET wii)
elseif(WIIU)
  set(OMORI_TARGET wiiu)
elseif(VITASDK)
  set(OMORI_TARGET vita)
elseif(PS3)
  set(OMORI_TARGET ps3)
elseif(PS2)
  set(OMORI_TARGET ps2)
elseif(XENON)
  set(OMORI_TARGET xenon)
else()
  set(OMORI_TARGET psp CACHE STRING "Console whose screen layout the desktop build reproduces (psp, vita, 3ds, wii, wiiu, ps3, ps2 or xenon)")
endif()
set(PROFILE_DIR ${CMAKE_SOURCE_DIR}/src/platform/${OMORI_TARGET})

if(PSP)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/psp/*.cpp src/platform/common/*.cpp src/platform/stdio/*.cpp)
  file(GLOB SOFTFLOAT_SRC CONFIGURE_DEPENDS third_party/compiler-rt-builtins/*.c)
  add_library(softfloat OBJECT ${SOFTFLOAT_SRC})
  target_compile_options(softfloat PRIVATE -O2 -G0 -fno-builtin -w)
  add_executable(omori ${ENGINE_SRC} ${PLATFORM_SRC} $<TARGET_OBJECTS:softfloat>)
  target_compile_options(omori PRIVATE -Os -G0 -ffunction-sections -fdata-sections -fno-strict-aliasing -Wall -Wno-unused-function -fexceptions)
  target_link_options(omori PRIVATE -Wl,--gc-sections)
  file(GLOB PSP_HOT_SRC CONFIGURE_DEPENDS src/gfx2d/*.cpp src/platform/psp/*.cpp src/core/image.cpp src/core/audio.cpp src/js/vm.cpp)
  set_source_files_properties(${PSP_HOT_SRC} PROPERTIES COMPILE_OPTIONS -O2)
  set_source_files_properties(src/omori/fx_core.cpp src/omori/fx_filters.cpp src/omori/fx_filters2.cpp PROPERTIES COMPILE_OPTIONS "-O2;-fsingle-precision-constant")
  option(USE_TREMOR "Decode OGG with Tremor instead of libvorbis" OFF)
  if(USE_TREMOR)
    target_compile_definitions(omori PRIVATE USE_TREMOR)
    set(VORBIS_LIBS vorbisidec ogg)
  else()
    set(VORBIS_LIBS vorbisfile vorbis ogg)
  endif()
  target_link_libraries(omori PRIVATE ${VORBIS_LIBS} z pspgu pspge pspdisplay pspctrl pspaudio psppower psprtc pspvfpu pspdebug pspjpeg psputility m)
  create_pbp_file(TARGET omori TITLE "OMORI" VERSION "01.00" BUILD_PRX)
  add_dependencies(omori build_info)
  target_include_directories(omori PRIVATE ${BUILD_INFO_DIR} third_party ${CMAKE_SOURCE_DIR}/../shared ${PROFILE_DIR})
elseif(NINTENDO_3DS)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/3ds/*.cpp src/platform/common/*.cpp src/platform/stdio/*.cpp)
  ctr_add_shader_library(omori_shader OUTPUT sprite.shbin src/platform/3ds/sprite.v.pica)
  dkp_add_embedded_binary_library(omori_shader_blob omori_shader)
  add_executable(omori ${ENGINE_SRC} ${PLATFORM_SRC})
  target_compile_options(omori PRIVATE -O2 -fno-strict-aliasing -Wall -Wno-unused-function -Wno-psabi -fexceptions)
  set_source_files_properties(src/omori/fx_core.cpp src/omori/fx_filters.cpp src/omori/fx_filters2.cpp PROPERTIES COMPILE_OPTIONS -fsingle-precision-constant)
  target_compile_definitions(omori PRIVATE USE_TREMOR)
  target_link_libraries(omori PRIVATE omori_shader_blob turbojpeg vorbisidec ogg z citro3d ctru m)
  add_dependencies(omori build_info)
  target_include_directories(omori PRIVATE ${BUILD_INFO_DIR} third_party ${CMAKE_SOURCE_DIR}/../shared ${DEVKITPRO}/portlibs/3ds/include ${PROFILE_DIR})
  ctr_create_3dsx(omori NOSMDH)
elseif(NINTENDO_WII)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/wii/*.cpp src/platform/common/*.cpp)
  add_executable(omori ${ENGINE_SRC} ${PLATFORM_SRC})
  target_compile_options(omori PRIVATE -O2 -fno-strict-aliasing -Wall -Wno-unused-function -fexceptions)
  target_link_libraries(omori PRIVATE turbojpeg vorbisfile vorbis ogg z wiiuse bte asnd fat ogc m)
  add_dependencies(omori build_info)
  target_include_directories(omori PRIVATE ${BUILD_INFO_DIR} third_party ${CMAKE_SOURCE_DIR}/../shared ${DEVKITPRO}/portlibs/ppc/include ${PROFILE_DIR})
  ogc_create_dol(omori)
elseif(WIIU)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/wiiu/*.cpp src/platform/common/*.cpp)
  file(GLOB WIIU_SHADERS CONFIGURE_DEPENDS src/platform/wiiu/shaders/*.gsh)
  set(WIIU_SHADERS_SRC ${BUILD_INFO_DIR}/wiiu_shaders.cpp)
  string(REPLACE ";" "|" WIIU_SHADERS_ARG "${WIIU_SHADERS}")
  add_custom_command(OUTPUT ${WIIU_SHADERS_SRC}
    COMMAND ${CMAKE_COMMAND} "-DIN=${WIIU_SHADERS_ARG}" -DOUT=${WIIU_SHADERS_SRC} -P ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake
    DEPENDS ${WIIU_SHADERS} ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake VERBATIM)
  add_executable(omori ${ENGINE_SRC} ${PLATFORM_SRC} ${WIIU_SHADERS_SRC})
  target_compile_options(omori PRIVATE -O2 -fno-strict-aliasing -fno-if-conversion -fno-if-conversion2 -Wall -Wno-unused-function -fexceptions)
  target_link_libraries(omori PRIVATE turbojpeg vorbisfile vorbis ogg z m)
  add_dependencies(omori build_info)
  target_include_directories(omori PRIVATE ${BUILD_INFO_DIR} third_party ${CMAKE_SOURCE_DIR}/../shared ${DEVKITPRO}/portlibs/ppc/include ${PROFILE_DIR} ${CMAKE_SOURCE_DIR}/src/platform)
  wut_create_rpx(omori)
elseif(VITASDK)
  include("${VITASDK}/share/vita.cmake" REQUIRED)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/vita/*.cpp src/platform/common/*.cpp)
  file(GLOB VITA_GXP CONFIGURE_DEPENDS src/platform/vita/shaders/*.gxp)
  set(VITA_SHADERS_SRC ${BUILD_INFO_DIR}/vita_shaders.cpp)
  string(REPLACE ";" "|" VITA_GXP_ARG "${VITA_GXP}")
  add_custom_command(OUTPUT ${VITA_SHADERS_SRC}
    COMMAND ${CMAKE_COMMAND} "-DIN=${VITA_GXP_ARG}" -DOUT=${VITA_SHADERS_SRC} -P ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake
    DEPENDS ${VITA_GXP} ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake VERBATIM)
  add_executable(omori ${ENGINE_SRC} ${PLATFORM_SRC} ${VITA_SHADERS_SRC})
  target_compile_options(omori PRIVATE -O2 -fno-strict-aliasing -Wall -Wno-unused-function -fexceptions)
  target_link_libraries(omori PRIVATE vorbisfile vorbis ogg z m SceGxm_stub SceDisplay_stub SceCtrl_stub SceAudio_stub SceAppUtil_stub SceAppMgr_stub
    SceSysmodule_stub ScePower_stub SceLibKernel_stub SceKernelThreadMgr_stub SceSysmem_stub SceIofilemgr_stub SceProcessmgr_stub)
  add_dependencies(omori build_info)
  target_include_directories(omori PRIVATE ${BUILD_INFO_DIR} third_party ${CMAKE_SOURCE_DIR}/../shared ${PROFILE_DIR} ${CMAKE_SOURCE_DIR}/src/platform)
  vita_create_self(eboot.bin omori UNSAFE)
  add_custom_command(OUTPUT param.sfo COMMAND vita-mksfoex -s TITLE_ID=OMRI00001 -s APP_VER=01.00 -s CATEGORY=gd -d ATTRIBUTE2=12 OMORI param.sfo)
  add_custom_target(param_sfo ALL DEPENDS param.sfo)
elseif(PS3)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/ps3/*.cpp src/platform/common/*.cpp)
  file(GLOB PS3_SHADERS CONFIGURE_DEPENDS src/platform/ps3/shaders/*.vpo src/platform/ps3/shaders/*.fpo)
  set(PS3_SHADERS_SRC ${BUILD_INFO_DIR}/ps3_shaders.cpp)
  string(REPLACE ";" "|" PS3_SHADERS_ARG "${PS3_SHADERS}")
  add_custom_command(OUTPUT ${PS3_SHADERS_SRC}
    COMMAND ${CMAKE_COMMAND} "-DIN=${PS3_SHADERS_ARG}" -DOUT=${PS3_SHADERS_SRC} -P ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake
    DEPENDS ${PS3_SHADERS} ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake VERBATIM)
  add_executable(omori ${ENGINE_SRC} ${PLATFORM_SRC} ${PS3_SHADERS_SRC})
  set_target_properties(omori PROPERTIES SUFFIX .elf)
  target_compile_options(omori PRIVATE -O2 -fno-strict-aliasing -Wall -Wno-unused-function -fexceptions
    "$<$<COMPILE_LANGUAGE:CXX>:SHELL:-include ${CMAKE_SOURCE_DIR}/src/platform/ps3/math_compat.h>")
  target_link_libraries(omori PRIVATE vorbisfile vorbis ogg z rsx gcm_sys io sysutil audio rt lv2 m)
  add_dependencies(omori build_info)
  target_include_directories(omori PRIVATE ${BUILD_INFO_DIR} third_party ${CMAKE_SOURCE_DIR}/../shared ${PROFILE_DIR} ${CMAKE_SOURCE_DIR}/src/platform)
  set(PS3_CONTENT_ID UP0001-OMRI00001_00-0000000000000000)
  add_custom_command(TARGET omori POST_BUILD
    COMMAND ${CMAKE_STRIP} $<TARGET_FILE:omori> -o ${CMAKE_BINARY_DIR}/omori.stripped.elf
    COMMAND ${PS3DEV}/bin/sprxlinker ${CMAKE_BINARY_DIR}/omori.stripped.elf
    COMMAND ${PS3DEV}/bin/make_self_npdrm ${CMAKE_BINARY_DIR}/omori.stripped.elf ${CMAKE_BINARY_DIR}/EBOOT.BIN ${PS3_CONTENT_ID}
    VERBATIM)
elseif(PS2)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/ps2/*.cpp src/platform/common/*.cpp)
  set(IRX_DIR ${PS2SDK}/iop/irx)
  set(IRX_SRC)
  foreach(m iomanX fileXio libsd audsrv usbd bdm bdmfs_fatfs usbmass_bd cdfs)
    add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/irx/${m}_irx.c
      COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/irx
      COMMAND ${PS2SDK}/bin/bin2c ${IRX_DIR}/${m}.irx ${CMAKE_BINARY_DIR}/irx/${m}_irx.c ${m}_irx
      DEPENDS ${IRX_DIR}/${m}.irx VERBATIM)
    list(APPEND IRX_SRC ${CMAKE_BINARY_DIR}/irx/${m}_irx.c)
  endforeach()
  file(GLOB SOFTFLOAT_SRC CONFIGURE_DEPENDS third_party/compiler-rt-builtins/*.c)
  add_library(softfloat OBJECT ${SOFTFLOAT_SRC})
  target_compile_options(softfloat PRIVATE -O2 -G0 -fno-builtin -w)
  add_executable(omori ${ENGINE_SRC} ${PLATFORM_SRC} ${IRX_SRC} $<TARGET_OBJECTS:softfloat>)
  set_target_properties(omori PROPERTIES SUFFIX .elf)
  target_compile_options(omori PRIVATE -Os -G0 -ffunction-sections -fdata-sections -fno-strict-aliasing -Wall -Wno-unused-function -fexceptions)
  string(REPLACE "-T$ENV{PS2SDK}/ee/startup/linkfile" "-T${CMAKE_SOURCE_DIR}/cmake/ps2.ld" CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS}")
  set_property(TARGET omori APPEND PROPERTY LINK_DEPENDS ${CMAKE_SOURCE_DIR}/cmake/ps2.ld)
  target_link_options(omori PRIVATE -Wl,--gc-sections)
  foreach(f Create Delete Post Pend CancellablePend)
    target_link_options(omori PRIVATE -Wl,--wrap=pte_osSemaphore${f})
  endforeach()
  set_source_files_properties(src/omori/fx_core.cpp src/omori/fx_filters.cpp src/omori/fx_filters2.cpp PROPERTIES COMPILE_OPTIONS -fsingle-precision-constant)
  target_compile_definitions(omori PRIVATE USE_TREMOR)
  target_link_libraries(omori PRIVATE turbojpeg vorbisidec ogg z audsrv pad cdvd patches fileXio graph dma -ldebug kernel m)
  add_dependencies(omori build_info)
  target_include_directories(omori PRIVATE ${BUILD_INFO_DIR} third_party ${CMAKE_SOURCE_DIR}/../shared ${PROFILE_DIR})
  add_custom_command(TARGET omori POST_BUILD
    COMMAND ${PS2DEV}/ee/bin/mips64r5900el-ps2-elf-strip --strip-all $<TARGET_FILE:omori> -o ${CMAKE_BINARY_DIR}/OMORI.ELF
    VERBATIM)
elseif(XENON)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/xenon/*.cpp src/platform/common/*.cpp)
  file(GLOB XENON_SHADERS CONFIGURE_DEPENDS src/platform/xenon/shaders/*.vsu src/platform/xenon/shaders/*.psu)
  set(XENON_SHADERS_SRC ${BUILD_INFO_DIR}/xenon_shaders.cpp)
  string(REPLACE ";" "|" XENON_SHADERS_ARG "${XENON_SHADERS}")
  add_custom_command(OUTPUT ${XENON_SHADERS_SRC}
    COMMAND ${CMAKE_COMMAND} "-DIN=${XENON_SHADERS_ARG}" -DOUT=${XENON_SHADERS_SRC} -P ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake
    DEPENDS ${XENON_SHADERS} ${CMAKE_SOURCE_DIR}/cmake/embed_shaders.cmake VERBATIM)
  add_executable(omori ${ENGINE_SRC} ${PLATFORM_SRC} ${XENON_SHADERS_SRC})
  set_target_properties(omori PROPERTIES SUFFIX .elf)
  target_compile_options(omori PRIVATE -O2 -fno-strict-aliasing -Wall -Wno-unused-function)
  foreach(f open read write lseek close fstat stat mkdir unlink rename ftruncate fsync opendir readdir closedir)
    target_link_options(omori PRIVATE -Wl,--wrap=${f})
  endforeach()
  target_link_libraries(omori PRIVATE -Wl,--start-group vorbisfile vorbis ogg z fat xenon m -Wl,--end-group)
  add_dependencies(omori build_info)
  target_include_directories(omori PRIVATE ${BUILD_INFO_DIR} third_party ${CMAKE_SOURCE_DIR}/../shared ${PROFILE_DIR} ${CMAKE_SOURCE_DIR}/src/platform)
  add_custom_command(TARGET omori POST_BUILD
    COMMAND ${CMAKE_STRIP} $<TARGET_FILE:omori> -o ${CMAKE_BINARY_DIR}/omori.stripped.elf
    COMMAND ${CMAKE_OBJCOPY} -O elf32-powerpc --adjust-vma 0x80000000 ${CMAKE_BINARY_DIR}/omori.stripped.elf ${CMAKE_BINARY_DIR}/xenon.elf
    VERBATIM)
else()
  find_package(SDL2 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(VORBIS REQUIRED vorbisfile)
  find_package(ZLIB REQUIRED)
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/desktop/*.cpp src/platform/common/*.cpp src/platform/stdio/*.cpp)
  add_executable(omori ${ENGINE_SRC} ${PLATFORM_SRC})
  target_include_directories(omori PRIVATE third_party ${VORBIS_INCLUDE_DIRS} ${BUILD_INFO_DIR} ${CMAKE_SOURCE_DIR}/../shared ${PROFILE_DIR})
  add_dependencies(omori build_info)
  target_link_libraries(omori PRIVATE SDL2::SDL2 ${VORBIS_LIBRARIES} ZLIB::ZLIB)
  target_compile_options(omori PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
  if(SANITIZE)
    target_compile_options(omori PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(omori PRIVATE -fsanitize=address,undefined)
  endif()
endif()
